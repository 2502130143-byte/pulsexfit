class _PulseXAppState extends State<PulseXApp> with WidgetsBindingObserver {
  @override

    super.initState();
    WidgetsBinding.instance.addObserver(this);
  }

  @override
  void dispose() {
    WidgetsBinding.instance.removeObserver(this);
    super.dispose();
  }

  // Modo ahorro: pausa el conteo de pasos en segundo plano.
  @override
  void didChangeAppLifecycleState(AppLifecycleState st) {

  }

  Widget build(BuildContext c) => MaterialApp(
    debugShowCheckedModeBanner: false,
    title: 'PulseX',
    theme: ThemeData.dark().copyWith(scaffoldBackgroundColor: bg),
    home: const SplashPage(),
  );

class LiveMap extends StatefulWidget {
  final List<LatLng> route;
  final LatLng? pos;
  final MapController? ctl;
  const LiveMap({super.key, this.route = const [], this.pos, this.ctl});
  @override
  State<LiveMap> createState() => _LiveMapState();
}

class _LiveMapState extends State<LiveMap> {
  late final MapController c = widget.ctl ?? MapController();
  LatLng? get p => widget.pos ?? (widget.route.isEmpty ? null : widget.route.last);

  @override
  void didUpdateWidget(LiveMap o) {
    super.didUpdateWidget(o);
    if (p != null && p != (o.pos ?? (o.route.isEmpty ? null : o.route.last))) {
      try {
        c.move(p!, c.camera.zoom);
      } catch (_) {}
    }
  }

  @override
  Widget build(BuildContext ctx) {
    final fit = widget.pos == null && widget.route.length > 1;
    return ClipRRect(
      borderRadius: BorderRadius.circular(18),
      child: FlutterMap(
        mapController: c,
        options: MapOptions(
          initialCenter: p ?? widget.route.firstOrNull ?? const LatLng(25.6736, -100.3245),
          initialZoom: 16,
          initialCameraFit: fit ? CameraFit.coordinates(coordinates: widget.route, padding: const EdgeInsets.all(30)) : null,
        ),
        children: [
          TileLayer(urlTemplate: 'https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}', userAgentPackageName: 'com.example.pulsex'),
          PolylineLayer(polylines: [Polyline(points: widget.route, strokeWidth: 5, color: cyan)]),
          if (p != null)
            MarkerLayer(markers: [
              Marker(point: p!, width: 26, height: 26, child: Container(decoration: BoxDecoration(color: blue, shape: BoxShape.circle, border: Border.all(color: Colors.white, width: 3)))),
            ]),
        ],
      ),
    );
  }
}

class LinePainter extends CustomPainter {
  final List<double> v;
  final Color col;
  LinePainter(this.v, this.col);
  @override
  void paint(Canvas cv, Size s) {
    if (v.length < 2) return;
    final p = Path();
    for (var i = 0; i < v.length; i++) {
      final x = s.width * i / (v.length - 1), y = s.height * (1 - v[i].clamp(0.0, 1.0));
      i == 0 ? p.moveTo(x, y) : p.lineTo(x, y);
    }
    cv.drawPath(p, Paint()..color = col..style = PaintingStyle.stroke..strokeWidth = 2);
  }

  @override
  bool shouldRepaint(_) => true;
}

class RoutePainter extends CustomPainter {
  final List<LatLng> r;
  RoutePainter(this.r);
  @override
  void paint(Canvas cv, Size s) {
    cv.drawRect(Offset.zero & s, Paint()..color = const Color(0xFF14301F));
    if (r.length < 2) return;
    final la = r.map((e) => e.latitude), lo = r.map((e) => e.longitude);
    final minLa = la.reduce(m.min), maxLa = la.reduce(m.max), minLo = lo.reduce(m.min), maxLo = lo.reduce(m.max);
    final dx = (maxLo - minLo) == 0 ? 1 : maxLo - minLo, dy = (maxLa - minLa) == 0 ? 1 : maxLa - minLa;
    final p = Path();
    for (var i = 0; i < r.length; i++) {
      final x = 6 + (s.width - 12) * (r[i].longitude - minLo) / dx, y = 6 + (s.height - 12) * (1 - (r[i].latitude - minLa) / dy);
      i == 0 ? p.moveTo(x, y) : p.lineTo(x, y);
    }
    cv.drawPath(p, Paint()..color = cyan..style = PaintingStyle.stroke..strokeWidth = 2);
  }

  @override
  bool shouldRepaint(_) => true;
}

class GaugePainter extends CustomPainter {
  final double value;
  GaugePainter(this.value);
  @override
  void paint(Canvas cv, Size s) {
    final r = Rect.fromCircle(center: Offset(s.width / 2, s.height - 10), radius: s.height - 20);
    cv.drawArc(r, m.pi, m.pi, false, Paint()..style = PaintingStyle.stroke..strokeWidth = 14..strokeCap = StrokeCap.round
      ..shader = const SweepGradient(startAngle: m.pi, endAngle: 2 * m.pi, colors: [Colors.teal, Colors.green, Colors.yellow, Colors.orange, Colors.red]).createShader(r));
    final a = m.pi + m.pi * value.clamp(0.0, 1.0);
    final c = r.center, rad = r.width / 2;
    cv.drawCircle(Offset(c.dx + rad * m.cos(a), c.dy + rad * m.sin(a)), 9, Paint()..color = Colors.white);
  }

  @override
  bool shouldRepaint(_) => true;
}

Future<void> credsDialog(BuildContext c, {bool reset = false}) {
  final e = TextEditingController(), p = TextEditingController(), p2 = TextEditingController();
  return showDialog(
    context: c,
    builder: (d) => AlertDialog(
      title: Text(reset ? 'Restablecer contraseña' : 'Crear cuenta'),
      content: Column(mainAxisSize: MainAxisSize.min, children: [
        TextField(controller: e, decoration: const InputDecoration(labelText: 'Correo electrónico'), keyboardType: TextInputType.emailAddress),
        if (!reset) TextField(controller: p, obscureText: true, decoration: const InputDecoration(labelText: 'Contraseña')),
        if (!reset) TextField(controller: p2, obscureText: true, decoration: const InputDecoration(labelText: 'Confirmar contraseña')),
      ]),
      actions: [
        TextButton(onPressed: () => Navigator.pop(d), child: const Text('Cancelar')),
        TextButton(
          onPressed: () async {
            final mail = e.text.trim();
            if (!RegExp(r'^[^@\s]+@[^@\s]+\.[^@\s]+$').hasMatch(mail)) return snack(c, 'Correo no válido');
            if (!reset) {
              if (p.text.length < 6) return snack(c, 'Mínimo 6 caracteres');
              if (p.text != p2.text) return snack(c, 'Las contraseñas no coinciden');
            }
            final err = reset ? await S.resetPass(mail) : await S.register(mail, p.text);
            if (!c.mounted) return;
            if (err != null) return snack(c, err);
            Navigator.pop(d);
            if (reset) {
              snack(c, 'Te enviamos un correo para restablecer tu contraseña');
            } else {
              snack(c, 'Cuenta creada. ¡Bienvenido!');
              go(c, const FingerprintPage());
            }
          },
          child: const Text('Aceptar'),
        ),
      ],
    ),
  );
}

// ---------- Acceso seguro (huella real + PIN) ----------
class FingerprintPage extends StatefulWidget {
  const FingerprintPage({super.key});
  @override
  State<FingerprintPage> createState() => _FingerprintPageState();
}

class _FingerprintPageState extends State<FingerprintPage> {
  String msg = '';
  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addPostFrameCallback((_) => auth());
  }

  Future<void> auth() async {
    if (await S.biometric()) {
      if (mounted) goRoot(context, const Shell());
    } else if (mounted) {
      setState(() => msg = 'No se pudo verificar la huella. Intenta de nuevo o usa tu PIN');
    }
  }

  Future<void> pin() async {
    final ctl = TextEditingController();
    final create = !S.hasPin;
    final ok = await showDialog<bool>(
      context: context,
      builder: (d) => AlertDialog(
        title: Text(create ? 'Crea tu PIN (4 dígitos)' : 'Ingresa tu PIN'),
        content: TextField(controller: ctl, obscureText: true, keyboardType: TextInputType.number, maxLength: 4, autofocus: true),
        actions: [
          TextButton(onPressed: () => Navigator.pop(d, false), child: const Text('Cancelar')),
          TextButton(
            onPressed: () {
              if (ctl.text.length != 4) return;
              if (create) {
                S.setPin(ctl.text);
                Navigator.pop(d, true);
              } else {
                Navigator.pop(d, S.checkPin(ctl.text));
              }
            },
            child: const Text('Aceptar'),
          ),
        ],
      ),
    );
    if (!mounted) return;
    if (ok == true) {
      goRoot(context, const Shell());
    } else if (ok == false && !create) {
      setState(() => msg = 'PIN incorrecto');
    }
  }

  @override
  Widget build(BuildContext c) => Scaffold(
    appBar: AppBar(backgroundColor: Colors.transparent, title: const Text('Acceso Seguro', style: TextStyle(fontSize: 16)), centerTitle: true),
    body: Bg(
      child: Center(
        child: Column(mainAxisAlignment: MainAxisAlignment.center, children: [
          neon(r: 24, child: glowIcon(Icons.fingerprint, 150)),
          const SizedBox(height: 30),
          const Text('Coloca tu huella dactilar\npara continuar', textAlign: TextAlign.center, style: TextStyle(fontSize: 17)),
          const SizedBox(height: 24),
          GestureDetector(onTap: auth, child: Container(padding: const EdgeInsets.all(18), decoration: BoxDecoration(shape: BoxShape.circle, border: Border.all(color: cyan, width: 3), boxShadow: [BoxShadow(color: cyan.withOpacity(.6), blurRadius: 30)]), child: const Icon(Icons.fingerprint, size: 60, color: cyan))),
          Padding(padding: const EdgeInsets.all(20), child: Text(msg, textAlign: TextAlign.center, style: const TextStyle(color: Colors.orangeAccent))),
          SizedBox(width: 200, child: OutlinedButton.icon(onPressed: pin, icon: const Icon(Icons.lock, size: 16), label: const Text('Usar PIN'))),
        ]),
      ),
    ),
  );
}

// ---------- Estadísticas / Perfil ----------
class StatsTab extends StatefulWidget {
  const StatsTab({super.key});
  @override
  State<StatsTab> createState() => _StatsTabState();
}

class _StatsTabState extends State<StatsTab> {
  int mode = 0;

  (List<String>, List<double>, List<Workout>) data() {
    final n = DateTime.now(), today = DateTime(n.year, n.month, n.day);
    final h = S.history;
    double sum(bool Function(Workout) t) => h.where(t).fold(0.0, (a, w) => a + w.km);
    DateTime d(Workout w) => DateTime(w.date.year, w.date.month, w.date.day);
    if (mode == 0) {
      final mon = today.subtract(Duration(days: today.weekday - 1));
      final vals = [for (var i = 0; i < 7; i++) sum((w) => d(w) == mon.add(Duration(days: i)))];
      return (['L', 'M', 'X', 'J', 'V', 'S', 'D'], vals, h.where((w) => !d(w).isBefore(mon)).toList());
    }
    if (mode == 1) {
      final vals = [for (var k = 3; k >= 0; k--) sum((w) => today.difference(d(w)).inDays ~/ 7 == k && !d(w).isAfter(today))];
      return (['S1', 'S2', 'S3', 'S4'], vals, h.where((w) => today.difference(d(w)).inDays < 28).toList());
    }
    final vals = [for (var k = 1; k <= 12; k++) sum((w) => w.date.year == n.year && w.date.month == k)];
    return (['E', 'F', 'M', 'A', 'M', 'J', 'J', 'A', 'S', 'O', 'N', 'D'], vals, h.where((w) => w.date.year == n.year).toList());
  }

  Widget badge(IconData i, Color col, String t, bool ok) => Opacity(opacity: ok ? 1 : .3, child: Column(children: [Icon(i, color: col, size: 40), Text(t, textAlign: TextAlign.center, style: const TextStyle(fontSize: 12))]));

  @override
  Widget build(BuildContext c) => ListenableBuilder(
    listenable: S,
    builder: (ctx, child) {
      final d = data();
      final labels = d.$1, vals = d.$2, ws = d.$3;
      final mx = vals.fold(0.1, m.max);
      final km = ws.fold(0.0, (a, w) => a + w.km), cal = ws.fold(0, (a, w) => a + w.cal), secs = ws.fold(0, (a, w) => a + w.secs);
      final days = S.history.map((w) => '${w.date.year}-${w.date.month}-${w.date.day}').toSet().length;
      final totalKm = S.history.fold(0.0, (a, w) => a + w.km);
      return ListView(padding: const EdgeInsets.all(16), children: [
        Row(children: [
          IconButton(icon: const Icon(Icons.settings), onPressed: () => go(c, const SettingsPage())),
          const Text('Tus Estadísticas', style: TextStyle(fontSize: 18)),
          const Spacer(),
          TextButton(onPressed: () => go(c, const SensorsPage()), child: const Text('Sensores', style: TextStyle(color: cyan))),
        ]),
        Wrap(spacing: 8, children: [for (final e in ['Semana', 'Mes', 'Año'].asMap().entries) ChoiceChip(label: Text(e.value), selected: mode == e.key, selectedColor: blue, onSelected: (_) => setState(() => mode = e.key))]),
        const SizedBox(height: 14),
        SizedBox(height: 140, child: Row(crossAxisAlignment: CrossAxisAlignment.end, mainAxisAlignment: MainAxisAlignment.spaceEvenly, children: [
          for (var i = 0; i < vals.length; i++)
            Column(mainAxisAlignment: MainAxisAlignment.end, children: [
              Container(width: 18, height: m.max(4, 110 * vals[i] / mx), decoration: BoxDecoration(borderRadius: BorderRadius.circular(4), gradient: const LinearGradient(begin: Alignment.bottomCenter, end: Alignment.topCenter, colors: [blue, purple]))),
              const SizedBox(height: 4),
              Text(labels[i], style: const TextStyle(fontSize: 12)),
            ]),
        ])),
        const SizedBox(height: 10),
        Row(mainAxisAlignment: MainAxisAlignment.spaceEvenly, children: [
          stat(Icons.directions_run, cyan, '${km.toStringAsFixed(1)} km', 'Distancia'),
          stat(Icons.local_fire_department, Colors.orange, num(cal), 'Calorías'),
          stat(Icons.timer, cyan, '${secs ~/ 3600}h ${two((secs % 3600) ~/ 60)}m', 'Tiempo'),
        ]),
        const SizedBox(height: 14),
        neon(child: Column(children: [
          const Row(children: [Icon(Icons.emoji_events, color: Colors.amber), SizedBox(width: 8), Text('Logros Desbloqueados')]),
          const SizedBox(height: 10),
          Row(mainAxisAlignment: MainAxisAlignment.spaceEvenly, children: [
            badge(Icons.workspace_premium, Colors.amber, 'Reto\n7 días', days >= 7),
            badge(Icons.hiking, cyan, '10K\nPasos', S.todaySteps >= 10000 || S.history.any((w) => w.steps >= 10000)),
            badge(Icons.star, Colors.amber, 'Maestro\ndel Movimiento', totalKm >= 50),
          ]),
        ])),
      ]);
    },
  );
}

// ---------- Sensores en tiempo real ----------
class SensorsPage extends StatefulWidget {
  const SensorsPage({super.key});
  @override
  State<SensorsPage> createState() => _SensorsPageState();
}

class _SensorsPageState extends State<SensorsPage> {
  List<double> a = [0, 0, 0], g = [0, 0, 0];
  double heading = 0;
  final ab = <double>[], gb = <double>[];
  final subs = <StreamSubscription>[];

  @override
  void initState() {
    super.initState();
    final iv = S.saver ? const Duration(seconds: 1) : SensorInterval.uiInterval;
    void push(List<double> b, double v) {
      b.add(v);
      if (b.length > 30) b.removeAt(0);
    }

    subs.add(accelerometerEventStream(samplingPeriod: iv).listen((e) => setState(() { a = [e.x, e.y, e.z]; push(ab, m.sqrt(e.x * e.x + e.y * e.y + e.z * e.z) / 20); }), onError: (_) {}));
    subs.add(gyroscopeEventStream(samplingPeriod: iv).listen((e) => setState(() { g = [e.x, e.y, e.z]; push(gb, m.sqrt(e.x * e.x + e.y * e.y + e.z * e.z) / 5); }), onError: (_) {}));
    final comp = FlutterCompass.events;
    if (comp != null) subs.add(comp.listen((e) => setState(() => heading = e.heading ?? heading), onError: (_) {}));
  }

  @override
  void dispose() {
    for (final s in subs) {
      s.cancel();
    }
    super.dispose();
  }

  Widget sensor(IconData i, Color col, String t, List<double> v, List<double> buf) => Padding(
    padding: const EdgeInsets.only(bottom: 12),
    child: neon(child: Row(children: [
      Icon(i, color: col, size: 44),
      const SizedBox(width: 12),
      Expanded(child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
        Text(t, style: const TextStyle(fontWeight: FontWeight.bold)),
        Text('X: ${v[0].toStringAsFixed(2)}', style: const TextStyle(fontSize: 13)),
        Text('Y: ${v[1].toStringAsFixed(2)}', style: const TextStyle(fontSize: 13)),
        Text('Z: ${v[2].toStringAsFixed(2)}', style: const TextStyle(fontSize: 13)),
      ])),
      SizedBox(width: 100, height: 60, child: CustomPaint(painter: LinePainter(buf, col))),
    ])),
  );

  @override
  Widget build(BuildContext c) {
    final h = (heading + 360) % 360;
    final dir = ['N', 'NE', 'E', 'SE', 'S', 'SO', 'O', 'NO'][((h + 22.5) ~/ 45) % 8];
    return Scaffold(
      appBar: AppBar(backgroundColor: Colors.transparent, title: const Text('Sensores en Tiempo Real', style: TextStyle(fontSize: 16)), actions: const [Center(child: Padding(padding: EdgeInsets.only(right: 12), child: Text('En vivo ●', style: TextStyle(color: Colors.greenAccent))))]),
      body: Bg(
        child: ListView(padding: const EdgeInsets.all(16), children: [
          sensor(Icons.open_with, cyan, 'Acelerómetro (m/s²)', a, ab),
          sensor(Icons.public, purple, 'Giroscopio (°/s)', g, gb),
          neon(child: Row(children: [
            Transform.rotate(angle: -h * m.pi / 180, child: const Icon(Icons.explore, size: 44, color: cyan)),
            const SizedBox(width: 12),
            const Text('Orientación'),
            const Spacer(),
            Text('$dir ${h.round()}°', style: const TextStyle(fontSize: 22, fontWeight: FontWeight.bold)),
          ])),
          const SizedBox(height: 12),
          neon(c: Colors.green, child: Row(children: [const Icon(Icons.check_circle, color: Colors.greenAccent), const SizedBox(width: 10), Text(S.saver ? 'Sensores activos\ncon bajo consumo (1 Hz)' : 'Sensores activos')])),
        ]),
      ),
    );
  }
}
